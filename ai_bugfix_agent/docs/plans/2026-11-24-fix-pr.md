# Stufe 3: Fix und Pull Request – Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ein dritter Schritt `fix` setzt den Plan um, prüft den grünen Test, pusht `bugfix/issue-N` und öffnet einen Pull Request gegen `TARGET_BASE`; alle Sitzungen arbeiten nur im Unterordner `TARGET_DIR` und ohne Geheimnisse in der Umgebung.

**Architecture:** `fix` ist ein `Step` wie Analyse und Plan. Push und Pull Request liegen hinter der Schnittstelle `Publisher`; die echte Umsetzung pusht mit dem Token über Git-Konfiguration aus Umgebungsvariablen und nutzt die GitHub-REST-API, Tests nutzen ein lokales Bare-Repo. Der Arbeitsbereich ergibt sich aus Worktree plus `TARGET_DIR`; die Policy prüft gegen ihn, weil er das `cwd` der Sitzung ist.

**Tech Stack:** TypeScript mit Node 22 `--experimental-strip-types`, `node:test`, Git ≥ 2.31 (für `GIT_CONFIG_COUNT`), GitHub-REST-API.

**Spec:** `docs/specs/2026-11-24-fix-pr-design.md`

```ts
function normalizeDir(targetDir: string): string {
  return targetDir.replaceAll('\\', '/').replace(/^\/+|\/+$/g, '')
}

/** Arbeitsbereich der Sitzung im Worktree; leeres `targetDir` = ganzer Worktree */
export function workdirOf(worktree: string, targetDir: string): string {
  const dir = normalizeDir(targetDir)
  return dir === '' ? worktree : join(worktree, dir)
}

/** Pfad relativ zum Arbeitsbereich → relativ zum Repo, mit `/` getrennt, wie `git status` ihn nennt */
export function toRepoPath(targetDir: string, path: string): string {
  const dir = normalizeDir(targetDir)
  const file = posix.normalize(path.replaceAll('\\', '/'))
  return dir === '' ? file : `${dir}/${file}`
}

/** Liegt ein Repo-Pfad im Arbeitsbereich? */
export function isUnder(targetDir: string, repoPath: string): boolean {
  const dir = normalizeDir(targetDir)
  return dir === '' || repoPath === dir || repoPath.startsWith(`${dir}/`)
}
```

```ts
/** Prüft selbst, was die Sitzung behauptet: genau die Testdatei geändert, Test rot, Name in der Ausgabe. */
async function verifyRedTest(
  worktree: string,
  workdir: string,
  targetDir: string,
  output: AnalyzeOutput,
  testCommand: string,
): Promise<Check> {
  if (output.testdatei === undefined || output.testname === undefined) {
    return { ok: false, reason: 'Das Ergebnis nennt keine Testdatei oder keinen Testnamen.' }
  }
  const file = relative(workdir, resolve(workdir, output.testdatei)).replaceAll('\\', '/')
  const repoFile = toRepoPath(targetDir, file)
  const changed = await changedFiles(worktree)
  if (changed.length !== 1 || changed[0] !== repoFile) {
    return { ok: false, reason: `Erwartet war nur eine Änderung an ${repoFile}, geändert: ${changed.join(', ') || 'nichts'}.` }
  }
  const run = await runShell(testCommand, workdir)
  if (run.code === 0) return { ok: false, reason: 'Mit dem neuen Test laufen alle Tests grün.' }
  if (!run.output.includes(output.testname)) {
    return { ok: false, reason: `Der Testname „${output.testname}“ kommt in der Testausgabe nicht vor.` }
  }
  return { ok: true, file, repoFile, testOutput: excerptFrom(run.output, output.testname) }
}
```

```ts
import { runFile } from '../command.ts'
import type { GitHubApi } from '../store/github.ts'

export type PullInput = { title: string; head: string; base: string; body: string }

/** Push und Pull Request. Echt über GitHub, in Tests gegen ein lokales Bare-Repo. */
export type Publisher = {
  /** Commit des Branches auf dem Remote, undefined wenn es ihn dort nicht gibt */
  remoteHead(branch: string): Promise<string | undefined>
  push(worktree: string, branch: string): Promise<void>
  findOpenPull(branch: string): Promise<string | undefined>
  createPull(input: PullInput): Promise<string>
}

function basicAuth(token: string): string {
  return Buffer.from(`x-access-token:${token}`).toString('base64')
}

/**
 * Der Token geht über Git-Konfiguration aus Umgebungsvariablen an Git, nicht
 * über die Kommandozeile. Der Credential-Helper bleibt aus, damit kein Dialog
 * aufgeht und keine gespeicherten Zugangsdaten des Rechners greifen.
 */
export function pushAuth(token: string): { args: string[]; env: NodeJS.ProcessEnv } {
  return {
    args: ['-c', 'credential.helper='],
    env: {
      ...process.env,
      GIT_TERMINAL_PROMPT: '0',
      GIT_CONFIG_COUNT: '1',
      GIT_CONFIG_KEY_0: 'http.https://github.com/.extraheader',
      GIT_CONFIG_VALUE_0: `Authorization: Basic ${basicAuth(token)}`,
    },
  }
}

export function redact(text: string, token: string): string {
  return text.replaceAll(token, '***').replaceAll(basicAuth(token), '***')
}

export function githubPublisher(
  token: string,
  githubRepo: string,
  pulls: Pick<GitHubApi, 'findOpenPull' | 'createPull'>,
  options: { url?: string } = {},
): Publisher {
  // Nicht `origin`: gepusht wird nur dorthin, wofür der Token gilt.
  const url = options.url ?? `https://github.com/${githubRepo}.git`
  const auth = pushAuth(token)

  async function gitWithToken(cwd: string, args: string[], what: string): Promise<string> {
    const result = await runFile('git', [...auth.args, ...args], cwd, auth.env)
    if (result.code !== 0) throw new Error(redact(`git ${what} fehlgeschlagen:\n${result.output.trim()}`, token))
    return result.stdout
  }

  return {
    async remoteHead(branch) {
      const out = await gitWithToken(process.cwd(), ['ls-remote', url, `refs/heads/${branch}`], 'ls-remote')
      return out.trim().split(/\s+/)[0] || undefined
    },
    async push(worktree, branch) {
      // Ohne --force: Weicht der Branch auf GitHub ab, scheitert der Push.
      await gitWithToken(worktree, ['push', url, `refs/heads/${branch}:refs/heads/${branch}`], 'push')
    },
    findOpenPull: (branch) => pulls.findOpenPull(branch),
    createPull: (input) => pulls.createPull(input),
  }
}
```

```ts
import { z } from 'zod'
import { runShell, tailLines } from '../command.ts'
import { addWorktree, branchExists, changedFiles, commitAll, git, removeWorktree } from '../git.ts'
import type { Ticket } from '../ticketfile.ts'
import {
  branchName,
  bulletList,
  firstSentence,
  isUnder,
  loadPrompt,
  outputSchema,
  quoteTicket,
  toRepoPath,
  workdirOf,
  worktreePath,
} from './common.ts'
import type { Publisher, PullInput } from './publish.ts'
import type { RunSession } from './session.ts'
import type { Step, StepResult } from './types.ts'

const PROMPT = loadPrompt('fix')
const MAX_TURNS = 40
const OUTPUT_LINES = 30

export const FixOutput = z.object({ zusammenfassung: z.string(), dateien: z.array(z.string()) })
type FixOutput = z.infer<typeof FixOutput>

const SCHEMA = outputSchema(FixOutput)

export type FixDeps = {
  repo: string
  /** Basis des Pull Requests */
  base: string
  testCommand: string
  worktreeRoot: string
  runSession: RunSession
  publish: Publisher
  /** Bezeichnung des Tickets für Titel und Text des Pull Requests, z. B. '#7' */
  label: (id: number) => string
  /** Arbeitsbereich relativ zum Repo; Default: das ganze Repo */
  targetDir?: string
  /** Default: bugfix/bug-N */
  branchName?: (id: number) => string
}

function planSummary(body: string): string {
  return /(?:^|\n)## Plan\n### Zusammenfassung\n([\s\S]*?)(?=\n#{2,3} |\s*$)/.exec(body)?.[1]?.trim() ?? '–'
}

function pullBody(label: string, fixSummary: string, plan: string, test: string): string {
  return [
    `Automatischer Fix des Bugfix-Agenten für ${label}.`,
    '',
    `Bezug: ${label}`,
    '',
    '## Fix',
    fixSummary,
    '',
    '## Plan',
    plan,
    '',
    '## Test',
    `\`${test}\``,
  ].join('\n')
}

function formatFix(output: FixOutput, changed: string[], url?: string, reason?: string): string {
  const parts = ['### Zusammenfassung', output.zusammenfassung.trim(), '', '### Geänderte Dateien', bulletList(changed)]
  if (url !== undefined) parts.push('', '### Pull Request', url)
  if (reason !== undefined) parts.push('', '### Prüfung gescheitert', reason)
  return parts.join('\n')
}

/** Prüft selbst, was die Sitzung behauptet. undefined, wenn alles passt, sonst der Grund. */
async function verifyFix(
  changed: string[],
  targetDir: string,
  testFile: string,
  testCommand: string,
  workdir: string,
): Promise<string | undefined> {
  if (changed.length === 0) return 'Die Sitzung hat nichts geändert.'
  const outside = changed.filter((file) => !isUnder(targetDir, file))
  if (outside.length > 0) return `Änderungen außerhalb von ${targetDir}: ${outside.join(', ')}`
  if (changed.includes(toRepoPath(targetDir, testFile))) return `Die Testdatei ${testFile} wurde geändert.`
  const run = await runShell(testCommand, workdir)
  if (run.code !== 0) return `Die Tests laufen nach dem Fix nicht grün:\n${tailLines(run.output, OUTPUT_LINES)}`
  return undefined
}

export function createFixStep(deps: FixDeps): Step {
  async function openPull(input: PullInput): Promise<string> {
    return (await deps.publish.findOpenPull(input.head)) ?? (await deps.publish.createPull(input))
  }

  return {
    name: 'fix',
    accepts: 'geplant',
    async run(ticket: Ticket): Promise<StepResult> {
      const id = ticket.id
      const label = deps.label(id)
      const title = ticket.header.get('titel') ?? ''
      const test = ticket.header.get('test') ?? ''
      const testFile = test.split(' › ')[0]?.trim() ?? ''
      if (testFile === '') throw new Error('Das Ticket hat kein Kopffeld test.')
      const branch = ticket.header.get('branch') ?? (deps.branchName ?? branchName)(id)
      if (!(await branchExists(deps.repo, branch))) {
        throw new Error(`Branch ${branch} fehlt. Status auf neu setzen, dann wird neu analysiert.`)
      }
      const pull = (fixSummary: string): PullInput => ({
        title: `Fix ${label}: ${title}`,
        head: branch,
        base: deps.base,
        body: pullBody(label, fixSummary, planSummary(ticket.body), test),
      })

      // Ein früherer Lauf hat gepusht, aber keinen PR angelegt: nur den PR nachholen.
      const local = (await git(deps.repo, ['rev-parse', branch])).trim()
      const subject = (await git(deps.repo, ['log', '-1', '--format=%s', branch])).trim()
      if (subject.startsWith(`bug ${id}: Fix`) && (await deps.publish.remoteHead(branch)) === local) {
        const url = await openPull(pull('Siehe Commit auf dem Branch.'))
        return {
          status: 'pr_offen',
          fields: { pr: url },
          section: { heading: 'Fix', text: `### Pull Request\n${url}` },
          summary: `PR nachgeholt: ${url}`,
        }
      }

      const worktree = worktreePath(deps.worktreeRoot, branch)
      await removeWorktree(deps.repo, worktree)
      await addWorktree(deps.repo, worktree, branch)
      try {
        const red = (
          await git(worktree, ['log', '--format=%H', '-n', '1', '--fixed-strings', `--grep=bug ${id}: roter Test`])
        ).trim()
        if (red === '') throw new Error(`Auf ${branch} fehlt der Commit „bug ${id}: roter Test“.`)
        // Reste eines früheren Versuchs fallen weg, z. B. ein nicht gepushter Fix-Commit.
        await git(worktree, ['reset', '-q', '--hard', red])

        const targetDir = deps.targetDir ?? ''
        const workdir = workdirOf(worktree, targetDir)
        const session = await deps.runSession({
          mode: 'fix',
          cwd: workdir,
          instructions: PROMPT,
          prompt: `Testkommando: ${deps.testCommand}\n\n${quoteTicket(ticket)}`,
          schema: SCHEMA,
          maxTurns: MAX_TURNS,
        })
        const output = FixOutput.parse(session.output)
        const cost = session.costUsd.toFixed(2)
        const changed = await changedFiles(worktree)
        const reason = await verifyFix(changed, targetDir, testFile, deps.testCommand, workdir)
        if (reason !== undefined) {
          return {
            status: 'fix_gescheitert',
            fields: { kosten_fix: cost },
            section: { heading: 'Fix', text: formatFix(output, changed, undefined, reason) },
            summary: `(${cost} $): ${firstSentence(reason)}`,
          }
        }

        await commitAll(worktree, `bug ${id}: Fix – ${title}`)
        await deps.publish.push(worktree, branch)
        const url = await openPull(pull(output.zusammenfassung.trim()))
        return {
          status: 'pr_offen',
          fields: { pr: url, kosten_fix: cost },
          section: { heading: 'Fix', text: formatFix(output, changed, url) },
          summary: `(${cost} $): ${firstSentence(output.zusammenfassung)} ${url}`,
        }
      } finally {
        await removeWorktree(deps.repo, worktree)
      }
    },
  }
}
```
